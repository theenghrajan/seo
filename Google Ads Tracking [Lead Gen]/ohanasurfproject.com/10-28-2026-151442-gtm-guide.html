2. Make "Call Today" buttons use the tracking number

The existing call tag only swaps the header and footer number. This makes all 9 call links dial Google's forwarding number.

Tag
Custom HTML (code below)
Trigger
Window Loaded, Page Path contains /lp
<script>
(function () {
  var tries = 0;
  (function wait() {
    if (typeof window._googWcmGet === 'function') {
      window._googWcmGet(function (formatted, mobile) {
        document.querySelectorAll('a[href^="tel:"]').forEach(function (a) {
          a.setAttribute('href', 'tel:' + mobile);
        });
      }, '808-809-9854');
    } else if (tries++ < 50) {
      setTimeout(wait, 200);
    }
  })();
})();
</script>

Test: open https://ohanasurfproject.com/lp-2/#google-wcc-debug in Preview, click Force in the grey box at the bottom left. Every call link, including all "Call Today" buttons, should change to tel:+9999999999.
