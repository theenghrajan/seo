## Tag 2 — SS - PPC Number Swap (Website) · Custom HTML · SS - Pageview (Website)
```html
<script>
(function () {
  var fromLink = /[?&]id=ppc(&|$)/.test(location.search);
  if (fromLink) document.cookie = 'ss_ppc=1; max-age=' + 60 * 60 * 24 * 30 + '; path=/; SameSite=Lax';
  if (!fromLink && document.cookie.indexOf('ss_ppc=1') === -1) return;

  var PPC = '808-809-9854';
  var TEXT = /\(?808\)?[\s.-]*599[\s.-]*(SURF\s*\(?7873\)?|7873)/gi;

  function swap() {
    document.querySelectorAll('a[href^="tel:"]').forEach(function (a) {
      if (a.getAttribute('href').replace(/\D/g, '').slice(-10) !== '8085997873') return;
      a.setAttribute('href', 'tel:8088099854');
      a.setAttribute('data-ss-ppc', '');
    });
    var walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, null, false), n;
    while ((n = walker.nextNode())) {
      if (TEXT.test(n.nodeValue)) n.nodeValue = n.nodeValue.replace(TEXT, PPC);
      TEXT.lastIndex = 0;
    }
    window.dataLayer.push({ event: 'ppcNumberSwapped' });

    // Google only rewrites links that show the number; copy its forwarding number to the rest (e.g. "Call Today" buttons)
    new MutationObserver(function () {
      var fwd = [].filter.call(document.querySelectorAll('a[data-ss-ppc]'), function (a) {
        return a.getAttribute('href') !== 'tel:8088099854';
      })[0];
      if (!fwd) return;
      document.querySelectorAll('a[data-ss-ppc][href="tel:8088099854"]').forEach(function (a) {
        a.setAttribute('href', fwd.getAttribute('href'));
      });
    }).observe(document.body, { subtree: true, attributes: true, attributeFilter: ['href'] });
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', swap); else swap();
})();
</script>
```
